cmake_minimum_required(VERSION 3.16)
project(native_desktop LANGUAGES CXX)
add_library(native_desktop STATIC
 project/src/Desktop.cpp project/src/MenuData.cpp project/src/Windows.cpp
 project/src/WindowsAppearance.cpp project/src/Linux.cpp)
target_include_directories(native_desktop PUBLIC project/include)
target_compile_features(native_desktop PUBLIC cxx_std_11)
find_package(Threads REQUIRED)
target_link_libraries(native_desktop PUBLIC Threads::Threads)
if(WIN32)
 target_link_libraries(native_desktop PUBLIC user32 shell32 gdi32 comctl32 advapi32)
elseif(APPLE)
 enable_language(OBJCXX)
 target_sources(native_desktop PRIVATE project/src/Mac.mm)
 target_link_libraries(native_desktop PUBLIC "-framework Cocoa")
elseif(UNIX)
 target_link_libraries(native_desktop PUBLIC ${CMAKE_DL_LIBS})
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(native_desktop_tests tests/NativeTests.cpp)
 target_link_libraries(native_desktop_tests PRIVATE native_desktop)
 add_test(NAME native_desktop COMMAND native_desktop_tests)
 set_tests_properties(native_desktop PROPERTIES TIMEOUT 10)
 if(WIN32)
  add_executable(native_desktop_appearance_tests tests/WindowsAppearanceTest.cpp)
  target_link_libraries(native_desktop_appearance_tests PRIVATE native_desktop)
  add_test(NAME windows_menu_appearance COMMAND native_desktop_appearance_tests)
  set_tests_properties(windows_menu_appearance PROPERTIES TIMEOUT 10)
 endif()
endif()
