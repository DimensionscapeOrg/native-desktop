cmake_minimum_required(VERSION 3.16)
project(native_desktop LANGUAGES CXX)
add_library(native_desktop STATIC
 project/src/ClipboardLinux.cpp project/src/ClipboardWindows.cpp project/src/Desktop.cpp project/src/DragDropLinux.cpp project/src/DragDropWindows.cpp project/src/MenuData.cpp project/src/Windows.cpp
 project/src/WindowsAppearance.cpp project/src/Linux.cpp project/src/WindowControlLinux.cpp project/src/WindowControlWindows.cpp project/src/Trash.cpp project/src/StorageVolumes.cpp)
target_include_directories(native_desktop PUBLIC project/include)
target_compile_features(native_desktop PUBLIC cxx_std_11)
find_package(Threads REQUIRED)
target_link_libraries(native_desktop PUBLIC Threads::Threads)
if(WIN32)
 target_link_libraries(native_desktop PUBLIC user32 shell32 gdi32 comctl32 advapi32 ole32)
elseif(APPLE)
 enable_language(OBJCXX)
 target_sources(native_desktop PRIVATE project/src/ClipboardMac.mm project/src/Mac.mm project/src/WindowControlMac.mm project/src/TrashMac.mm project/src/StorageVolumesMac.mm project/src/DragDropMac.mm)
 target_link_libraries(native_desktop PUBLIC "-framework Cocoa" "-framework Foundation")
elseif(UNIX)
 target_link_libraries(native_desktop PUBLIC ${CMAKE_DL_LIBS})
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(native_desktop_tests tests/NativeTests.cpp)
 target_link_libraries(native_desktop_tests PRIVATE native_desktop)
 add_test(NAME native_desktop COMMAND native_desktop_tests)
 set_tests_properties(native_desktop PROPERTIES TIMEOUT 10)
 if(APPLE)
  add_executable(native_desktop_mac_menu_tests tests/MacMenuTest.mm)
  target_link_libraries(native_desktop_mac_menu_tests PRIVATE native_desktop)
  add_test(NAME macos_menu_bar COMMAND native_desktop_mac_menu_tests)
  set_tests_properties(macos_menu_bar PROPERTIES TIMEOUT 10)
  add_executable(native_desktop_mac_window_tests tests/MacWindowTest.mm)
  target_link_libraries(native_desktop_mac_window_tests PRIVATE native_desktop)
  add_test(NAME macos_window_stacking COMMAND native_desktop_mac_window_tests)
  set_tests_properties(macos_window_stacking PROPERTIES TIMEOUT 10)
 endif()
 if(WIN32)
  add_executable(native_desktop_appearance_tests tests/WindowsAppearanceTest.cpp)
  target_link_libraries(native_desktop_appearance_tests PRIVATE native_desktop)
  add_test(NAME windows_menu_appearance COMMAND native_desktop_appearance_tests)
  set_tests_properties(windows_menu_appearance PROPERTIES TIMEOUT 10)
 endif()
endif()
